# Packages needed to BUILD the documentation.
#
# Used by Read the Docs, the GitHub Pages workflow, and local builds.
# Note: worktoy itself is NOT installed for the docs build — the docs
# only read the source files as text, never import the package.

# Sphinx builds the navigable, searchable, themed site.
sphinx>=7

# furo provides the HTML theme (html_theme in conf.py).
furo>=2024.1

# markwork generates the verbatim source pages (extensions in
# conf.py). Its host-facing contract is intentionally stable: it asks
# only that the sources live in 'src' and the tests in 'tests', so
# newer releases stay compatible and worktoy floats forward to pick up
# generator fixes. The floor is 0.1.0, the first stable release; a '>='
# floor also keeps pip from pulling pre-releases.
markwork>=0.1.0
